Table: courses
SQL: CREATE TABLE courses (
    id TEXT PRIMARY KEY,
    title TEXT NOT NULL DEFAULT '',
    from_language TEXT NOT NULL,
    learning_language TEXT NOT NULL
)

Table: sections
SQL: CREATE TABLE sections (
    id TEXT PRIMARY KEY,
    course_id TEXT NOT NULL,
    section_number INTEGER NOT NULL CHECK (section_number >= 0),
    title TEXT NOT NULL DEFAULT '',
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE
)

Table: units
SQL: CREATE TABLE units (
    id TEXT PRIMARY KEY,
    section_id TEXT NOT NULL,
    unit_number INTEGER NOT NULL CHECK (unit_number >= 0),
    unit_index INTEGER NOT NULL CHECK (unit_index >= 0),
    title TEXT NOT NULL DEFAULT '',
    FOREIGN KEY (section_id) REFERENCES sections(id) ON DELETE CASCADE
)

Table: levels
SQL: CREATE TABLE levels (
    id TEXT PRIMARY KEY,
    unit_id TEXT NOT NULL,
    absolute_node_index INTEGER CHECK (absolute_node_index >= 0),
    level_type TEXT NOT NULL,
    level_subtype TEXT NOT NULL,
    level_state TEXT NOT NULL,
    skill_id TEXT NOT NULL,
    tree_id TEXT NOT NULL,
    level_index INTEGER NOT NULL CHECK (level_index >= 0),
    total_sessions INTEGER NOT NULL DEFAULT 0 CHECK (total_sessions >= 0),
    teaching_objective TEXT NOT NULL DEFAULT '',
    debug_name TEXT,
    FOREIGN KEY (unit_id) REFERENCES units(id) ON DELETE CASCADE
)

Table: sessions
SQL: CREATE TABLE sessions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    level_id TEXT NOT NULL,
    level_session_index INTEGER NOT NULL CHECK (level_session_index >= 0),
    duolingo_session_id TEXT UNIQUE,
    file_name TEXT,
    UNIQUE(level_id, level_session_index),
    FOREIGN KEY (level_id) REFERENCES levels(id) ON DELETE CASCADE
)

Table: characters
SQL: CREATE TABLE characters (
    id TEXT PRIMARY KEY,
    name TEXT NOT NULL,
    gender TEXT CHECK(gender IN ('MALE', 'FEMALE', 'NEUTRAL', 'UNKNOWN')) NOT NULL DEFAULT 'UNKNOWN',
    url TEXT,
    pdf_url TEXT,
    svg_url TEXT,
    avatar_icon_pdf TEXT,
    avatar_icon_svg TEXT,
    lottie_correct TEXT,
    lottie_incorrect TEXT,
    lottie_idle TEXT
)

Table: challenges
SQL: CREATE TABLE challenges (
    id TEXT PRIMARY KEY,
    session_id INTEGER NOT NULL,
    type TEXT NOT NULL,
    prompt TEXT,
    tts TEXT,
    slow_tts TEXT,
    is_speaker_universal BOOLEAN NOT NULL DEFAULT 0 CHECK(is_speaker_universal IN (0, 1)),
    character_id TEXT,
    
    -- Cột Từ vựng (Tokens) dạng mảng JSON []
    tokens_text TEXT,    -- VD: ["我", "学"]
    tokens_tts TEXT,     -- VD: ["url1", "url2"]
    tokens_hints TEXT,   -- VD: ["tôi", "học"] (Mảng chuỗi)
    
    -- Cột Phương án (Choices) dạng mảng JSON []
    choices_text TEXT,   -- VD: ["canh", "trà"]
    choices_tts TEXT,
    choices_image TEXT,
    choices_correct TEXT, -- VD: [1, 0, 0, 0] (Mảng số nguyên 0/1)
    
    -- Cột Đáp án (Solutions) dạng mảng JSON []
    solutions TEXT,      -- VD: ["tôi học tiếng trung", "tớ học"]
    
    FOREIGN KEY (session_id) REFERENCES sessions(id) ON DELETE CASCADE,
    FOREIGN KEY (character_id) REFERENCES characters(id) ON DELETE SET NULL
)

Table: flashcards
SQL: CREATE TABLE flashcards (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        word TEXT UNIQUE NOT NULL,
        pinyin TEXT,
        meaning TEXT,
        tts_url TEXT
    )
